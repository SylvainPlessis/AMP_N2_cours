\centering%
\def\tor#1{\pgfmathparse{(3/4 * #1)^(1/3)}}
\tikzfading[name=fade inside,inner color=transparent!80, outer color=transparent!30]
\tikzfading[name=fade down,top color=transparent!80, bottom color=transparent!0]
\begin{tikzpicture}[y=3mm]
\fill[draw=none,path fading=fade down,fill=cyan!60] decorate[decoration=snake, segment length=20]{(0,0) -- (3.5,0)} -- ++(0,-23) -- (0,-23) -- cycle;
\foreach \y in {0,5,...,20}
{
  %% volume
  \pgfmathparse{600 / (1 + \y/10)}\let\v\pgfmathresult
  \pgfmathparse{600 - \v}\let\u\pgfmathresult
  \pgfmathparse{\u * (1 + \y/10)}\let\us\pgfmathresult
  %% rayon
  \pgfmathparse{(3/4 * \v)^(1/3)}\let\rv\pgfmathresult
  \pgfmathparse{(3/4 * \u)^(1/3)}\let\ru\pgfmathresult
  \pgfmathparse{(3/4 * \us)^(1/3)}\let\rus\pgfmathresult
  \draw (-2mm,-\y) -- ++(4mm,0) node[pos=0,anchor=east]{\y~m};
  \node[scale=3,inner sep=0pt] at (1,-\y) {\twemoji{lungs}};
  \ifnum\y=0
    \fill[path fading=fade inside, ball color=white,semitransparent] (1,-\y) circle (\rv pt);
  \else
    \fill[path fading=fade inside, ball color=white,semitransparent] (0.75,-\y) circle (\rv pt);
    \fill[path fading=fade inside, ball color=blue,semitransparent] (1.25,-\y) circle (\ru pt);
    \fill[path fading=fade inside, ball color=cyan] (2.5,-\y) circle (\rus pt);
  \fi
}
\end{tikzpicture}%
