\begin{itemize}
\item Équilibrer dès les premiers mètres (et même la surface) et régulièrement

\item privilégier les variations lentes de profondeur;
\item pas de Valsalva brutal;
\item si difficulté, remonter de quelques mètres, puis descendre lentement,
  tête en haut de préférence (ne pas forcer) \emph{préférer stopper la plongée
  plutôt que de risquer une lésion de l'oreille}.
\end{itemize}
\end{exampleblock}%
\begin{exampleblock}<only@4>{Comportement: à la remontée}
\begin{itemize}
\item Privilégier les variations lentes de profondeur;
\item pas de Valsava;
\item si difficulté, redescendre de quelques mètres, puis remonter lentement
  (ne pas forcer).
\end{itemize}
\end{exampleblock}%
\begin{exampleblock}<only@5>{Comportement: après la plongée}
\begin{itemize}
\item Rincer, sécher et protéger les oreilles;
\item éviter le froid, le vent\dots

\begin{block}<only@2>{Le mécanisme}
À la remontée, \\flux d'air sortant non suffisant $\Rightarrow$
les alvéoles atteignent leur limite d'élasticité:
\begin{itemize}
\item distension;
\item rupture;
\item passage de bulles directement dans le système sanguin.
\end{itemize}
\end{block}%
\begin{alertblock}<only@3>{Les symptômes}
\begin{itemize}
\item Douleurs dans la poitrine;
\item gêne respiratoire;
\item crachats \og spumeux\fg\ (traces de sang).
\end{itemize}
\end{alertblock}%
\begin{alertblock}<only@3>{Arrivée sur le bateau}
\begin{itemize}
\item État de choc;
\item convulsions;
\item troubles de la parole, de la vision, paralysie.
\end{itemize}
\end{alertblock}%
\begin{exampleblock}<only@4>{La prévention}
\begin{itemize}
\item \EMPH[red]{Ne jamais bloquer la respiration}, y  compris lors de remontée d’urgence sans embout;
\item ventiler normalement;
\item ne jamais donner d’air à un apnéiste non averti, y compris en piscine.
\end{itemize}
\end{exampleblock}%
\begin{alertblock}<only@5>{Si accident}
\begin{itemize}
\item Appeler les secours;
\item donner de l'\ce{O2};
\item hydrater.
\end{itemize}
\end{alertblock}%
